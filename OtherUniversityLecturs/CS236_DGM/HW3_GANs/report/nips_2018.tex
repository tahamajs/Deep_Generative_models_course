\section{Introduction}
Generative Adversarial Networks (GANs) formulate generative modelling as a two-player minimax game between a generator $G$ and a discriminator $D$. This assignment required implementing (i) a vanilla "non-saturating" GAN objective, (ii) a conditional GAN variant where class labels are provided to both $G$ and $D$, and (iii) an alternative Wasserstein objective with gradient penalty (WGAN-GP). The experiments use FashionMNIST (28x28 grayscale) as a standard benchmark.

\subsection{Problem 1: Generative Adversarial Networks (15 points)}

\subsubsection{Minimax loss and vanishing gradients [5 points]}
The original minimax objective for the discriminator and generator is:
\[
\min_G \max_D V(D,G)=\mathbb{E}_{x\sim p_{\text{data}}}[\log D(x)] + \mathbb{E}_{z\sim p_z}[\log(1-D(G(z)))] .
\]

In practice, training the generator by minimizing $\log(1-D(G(z)))$ leads to vanishing gradients early in training when $D$ becomes confident. The generator's loss is:
\[
L_{\text{minimax}}^G (\theta; \phi) = \mathbb{E}_{z\sim N(0,I)} [\log(1 - \sigma(h_\phi(G_\theta(z))))].
\]

\textbf{Derivation:} Using the chain rule and the fact that $\sigma'(x) = \sigma(x)(1 - \sigma(x))$,
\[
\frac{\partial L_{\text{minimax}}^G}{\partial \theta} = \mathbb{E}_{z\sim N(0,I)} \left[ \frac{\sigma'(h_\phi(G_\theta(z)))}{1 - \sigma(h_\phi(G_\theta(z)))} \cdot \frac{\partial h_\phi(G_\theta(z))}{\partial \theta} \right].
\]

Since $\sigma'(x) = \sigma(x)(1 - \sigma(x))$, we get:
\[
\frac{\partial L_{\text{minimax}}^G}{\partial \theta} = \mathbb{E}_{z\sim N(0,I)} \left[ \frac{\sigma(h_\phi(G_\theta(z))) \cdot (1 - \sigma(h_\phi(G_\theta(z))))}{1 - \sigma(h_\phi(G_\theta(z)))} \cdot \frac{\partial h_\phi(G_\theta(z))}{\partial \theta} \right] = \mathbb{E}_{z\sim N(0,I)} \left[ \sigma(h_\phi(G_\theta(z))) \cdot \frac{\partial h_\phi(G_\theta(z))}{\partial \theta} \right].
\]

When the discriminator successfully identifies fake samples (i.e., $\sigma(h_\phi(G_\theta(z))) \approx 0$), the gradient $\frac{\partial L_{\text{minimax}}^G}{\partial \theta} \approx 0$, causing training to slow or stop.

\subsubsection{Non-saturating GAN implementation [10 points]}
To address vanishing gradients, we use the non-saturating loss:
\[
L_G^{\text{non-saturating}}(\theta; \phi) = -\mathbb{E}_{z\sim N(0,I)} [\log \sigma(h_\phi(G_\theta(z)))].
\]

\textbf{Implementation:} The implementation uses Monte Carlo estimates:
\[
L_D(\phi; \theta) \approx -\frac{1}{m} \sum_{i=1}^m \log \sigma(h_\phi(x^{(i)})) - \frac{1}{m} \sum_{i=1}^m \log(1 - \sigma(h_\phi(G_\theta(z^{(i)}))))
\]
\[
L_G^{\text{non-saturating}}(\theta; \phi) \approx -\frac{1}{m} \sum_{i=1}^m \log \sigma(h_\phi(G_\theta(z^{(i)})))
\]

where $m$ is batch size, $x^{(i)} \sim p_{\text{data}}$, $z^{(i)} \sim N(0,I)$.

\subsection{Problem 2: Divergence Minimization (25 points)}

\subsubsection{Optimal discriminator characterization [10 points]}
Show that $L_D$ is minimized when $D_\phi = D^*$, where $D^*(x) = \frac{p_{\text{data}}(x)}{p_\theta(x) + p_{\text{data}}(x)}$.

\textbf{Derivation:} For fixed $x$, minimize $f(t) = -p_{\text{data}}(x) \log t - p_\theta(x) \log(1-t)$. Setting $f'(t) = 0$:
\[
-\frac{p_{\text{data}}(x)}{t} + \frac{p_\theta(x)}{1-t} = 0
\]
\[
p_\theta(x) t = p_{\text{data}}(x) (1-t)
\]
\[
t(p_\theta(x) + p_{\text{data}}(x)) = p_{\text{data}}(x)
\]
\[
t = \frac{p_{\text{data}}(x)}{p_\theta(x) + p_{\text{data}}(x)}
\]

Since $f$ is sum of strictly convex functions, this is the unique minimizer.

\subsubsection{Discriminator logits as log-likelihood ratios [5 points]}
When $D_\phi = D^* = \sigma(h_\phi(x))$, show that $h_\phi(x) = \log \frac{p_{\text{data}}(x)}{p_\theta(x)}$.

\textbf{Derivation:} $D^*(x) = \sigma(h_\phi(x)) = \frac{1}{1 + e^{-h_\phi(x)}} = \frac{p_{\text{data}}(x)}{p_\theta(x) + p_{\text{data}}(x)}$
\[
1 + e^{-h_\phi(x)} = \frac{p_\theta(x) + p_{\text{data}}(x)}{p_{\text{data}}(x)} = 1 + \frac{p_\theta(x)}{p_{\text{data}}(x)}
\]
\[
e^{-h_\phi(x)} = \frac{p_\theta(x)}{p_{\text{data}}(x)}
\]
\[
-h_\phi(x) = \log \frac{p_\theta(x)}{p_{\text{data}}(x)}
\]
\[
h_\phi(x) = \log \frac{p_{\text{data}}(x)}{p_\theta(x)}
\]

\subsubsection{Specific generator loss and KL divergence [5 points]}
Consider $L_G(\theta; \phi) = \mathbb{E}_{x\sim p_\theta(x)} [\log(1 - D_\phi(x))] - \mathbb{E}_{x\sim p_\theta(x)} [\log D_\phi(x)]$.
Show that if $D_\phi = D^*$, then $L_G(\theta; \phi) = KL(p_\theta(x)||p_{\text{data}}(x))$.

\textbf{Derivation:}
\[
L_G = \mathbb{E}_{p_\theta} [\log(1 - D^*)] - \mathbb{E}_{p_\theta} [\log D^*]
= \mathbb{E}_{p_\theta} \left[ \log \frac{1 - D^*}{D^*} \right]
= \mathbb{E}_{p_\theta} \left[ \log \frac{p_\theta}{p_{\text{data}}} \right]
= \mathbb{E}_{p_\theta} \left[ \log \frac{p_\theta(x)}{p_{\text{data}}(x)} \right] = KL(p_\theta||p_{\text{data}})
\]

\subsubsection{Comparing VAEs and GANs [5 points]}
Show that the negative log likelihood can be written as $KL(p_{\text{data}}||p_\theta) + C$ where $C$ is constant w.r.t. $\theta$.

The VAE minimizes $-ELBO \leq -\log p_\theta(x) + KL(q(z|x)||p(z))$. But the true objective is $KL(p_{\text{data}}||p_\theta) = \mathbb{E}_{p_{\text{data}}} [\log p_{\text{data}} - \log p_\theta] = C - \mathbb{E}_{p_{\text{data}}} [\log p_\theta]$ where $C = \mathbb{E}_{p_{\text{data}}} [\log p_{\text{data}}]$ is constant. However, $KL(p_\theta||p_{\text{data}}) \neq KL(p_{\text{data}}||p_\theta)$ in general, so VAEs and GANs optimize different objectives.

\subsection{Problem 3: Conditional GAN with Projection Discriminator (20 points)}

\subsubsection{Optimal conditional discriminator [10 points]}
Assume feature mapping $\phi(x)$ where both $p_{\text{data}}(x|y)$ and $p_\theta(x|y)$ are mixtures of Gaussians with means $\mu_y$ and $\hat{\mu}_y$. Show that optimal logits $h^*(x,y) = y^T (A \phi(x) + b)$.

\textbf{Derivation:} The log-likelihood ratio is:
\[
\log \frac{p_{\text{data}}(x,y)}{p_\theta(x,y)} = \log \frac{p_{\text{data}}(x|y)}{p_\theta(x|y)}
\]

For Gaussians: $\log \frac{\mathcal{N}(\phi(x)|\mu_y,I)}{\mathcal{N}(\phi(x)|\hat{\mu}_y,I)} = -\frac{1}{2}||\phi(x)-\mu_y||^2 + \frac{1}{2}||\phi(x)-\hat{\mu}_y||^2$

\[
= (\hat{\mu}_y - \mu_y)^T \phi(x) + \frac{1}{2}(||\hat{\mu}_y||^2 - ||\mu_y||^2)
\]

For one-hot $y$, this gives the form $y^T (A \phi(x) + b)$.

\subsubsection{Conditional GAN implementation [10 points]}
Implement conditional GAN with projection discriminator. The generator concatenates one-hot labels to latent vectors, discriminator uses projection form.

\subsection{Problem 4: Wasserstein GAN (35 points)}

\subsubsection{KL divergence for Gaussians [5 points]}
Let $p_\theta(x) = \mathcal{N}(x|\theta, \epsilon^2)$, $p_{\text{data}}(x) = \mathcal{N}(x|\theta_0, \epsilon^2)$. Show $KL(p_\theta||p_{\text{data}}) = \frac{(\theta - \theta_0)^2}{2\epsilon^2}$.

\textbf{Derivation:} Standard KL for Gaussians with same variance gives this result.

\subsubsection{Limit behavior and instability [5 points]}
As $\epsilon \to 0$, unless $\theta = \theta_0$, KL diverges and its gradients become extremely large, causing training instability.

\subsubsection{Problems with unconstrained Wasserstein critic [5 points]}
For $\epsilon \to 0$ (point masses), the critic loss $L_D = \mathbb{E}_{p_\theta}[f(x)] - \mathbb{E}_{p_{\text{data}}}[f(x)]$ can approach $-\infty$ by making $f(\theta) \to -\infty$ or $f(\theta_0) \to \infty$, so no optimal discriminator exists.

\subsubsection{Lipschitz-constrained critic [5 points]}
Restricting to 1-Lipschitz functions (derivative in [-1,1]), the optimal critic is $f(x) = c - |\theta - \theta_0|$ for $x$ between $\theta$ and $\theta_0$.

\subsubsection{WGAN-GP implementation [15 points]}
WGAN-GP adds gradient penalty: $L_D = \mathbb{E}_{p_\theta}[f] - \mathbb{E}_{p_{\text{data}}}[f] + \lambda \mathbb{E}_{p_{\hat{x}}} [(||\nabla f(x)||_2 - 1)^2]$ where $p_{\hat{x}}$ interpolates between real and fake samples.

\subsection{Problem 5: Noise Contrastive Estimation (5 points)}

\subsubsection{Optimal NCE parameters [5 points]}
Show that the optimal parameters satisfy $\frac{1}{Z^*} \exp(-E_{\theta^*}(x)) = p_{\text{data}}(x)$.

\textbf{Derivation:} Using the optimal discriminator result, the NCE objective leads to $E_{\theta^*}(x) + \log Z^* + \log p_{\text{noise}}(x) = \log \frac{p_{\text{data}}(x)}{p_{\text{noise}}(x)}$. Exponentiating gives the result.

\section{Implementation details}
\subsection{Architectures}
We replicate the exact layer choices used in the codebase (`src/codebase/network.py`). For clarity we restate them here:
\begin{itemize}
  \item Generator (dim_z = 64, for grayscale 28x28 images):
    \begin{itemize}
      \item Linear(dim_z, 512) + BatchNorm + ReLU
      \item Linear(512, 64*7*7) + BatchNorm + ReLU
      \item Reshape(64,7,7)
      \item PixelShuffle(2) followed by Conv2d(64//4, 32, k=3, p=1) + BatchNorm + ReLU
      \item PixelShuffle(2) followed by Conv2d(32//4, 1, k=3, p=1) (output logits scaled to [-1,1] with tanh or left as raw)
    \end{itemize}
  \item Discriminator:
    \begin{itemize}
      \item Conv2d(1,32,k=4,stride=2,p=1) + LeakyReLU(0.1)
      \item Conv2d(32,64,k=4,stride=2,p=1) + LeakyReLU(0.1)
      \item Flatten, Linear(64*7*7,512) + LeakyReLU(0.1)
      \item Linear(512,1) (logit output for standard GAN)
    \end{itemize}
  \item ConditionalGenerator / ConditionalDiscriminator: the conditional generator concatenates a one-hot label vector to the latent $z$ and otherwise follows the same upsampling pipeline. The conditional discriminator produces per-class logits and selects the logit corresponding to the given label (implementation detail: `gather` on the final linear output).
\end{itemize}

\subsection{Loss implementation notes and code}
The assignment's `src/codebase/gan.py` expects the following implementations:

\subsubsection{Non-saturating GAN loss}
\begin{verbatim}
def loss_nonsaturating(g, d, x_real, *, device):
    batch_size = x_real.shape[0]
    z = torch.randn(batch_size, g.dim_z, device=device)
    x_fake = g(z)
    d_real = d(x_real)
    d_fake = d(x_fake)

    # Either of the below are acceptable
    d_loss = -F.logsigmoid(d_real).mean() - F.logsigmoid(-d_fake).mean()
    d_loss = F.binary_cross_entropy_with_logits(d_real,
                torch.ones(batch_size, device=device)) + \
            F.binary_cross_entropy_with_logits(d_fake,
                torch.zeros(batch_size, device=device))

    g_loss = -F.logsigmoid(d_fake).mean()
    return d_loss, g_loss
\end{verbatim}

\subsubsection{Conditional non-saturating GAN loss}
\begin{verbatim}
def conditional_loss_nonsaturating(g, d, x_real, y_real, *, device):
    batch_size = x_real.shape[0]
    z = torch.randn(batch_size, g.dim_z, device=device)
    x_fake = g(z, y_real)  # Use same labels for fake samples
    d_real = d(x_real, y_real)
    d_fake = d(x_fake, y_real)

    d_loss = -F.logsigmoid(d_real).mean() - F.logsigmoid(-d_fake).mean()
    g_loss = -F.logsigmoid(d_fake).mean()
    return d_loss, g_loss
\end{verbatim}

\subsubsection{Wasserstein GAN with gradient penalty}
\begin{verbatim}
def loss_wasserstein_gp(g, d, x_real, *, device):
    batch_size = x_real.shape[0]
    z = torch.randn(batch_size, g.dim_z, device=device)
    x_fake = g(z)
    d_real = d(x_real)
    d_fake = d(x_fake)

    alpha = torch.rand(x_real.shape[0], 1, 1, 1, device=device)
    x_r = alpha * x_fake + (1 - alpha) * x_real
    d_r = d(x_r)

    grad = torch.autograd.grad(d_r.sum(), x_r, create_graph=True)
    grad_norm = grad[0].reshape(batch_size, -1).norm(dim=1)
    d_loss = (d_fake - d_real).mean() + 10 * ((grad_norm - 1)**2).mean()
    g_loss = -d_fake.mean()
    return d_loss, g_loss
\end{verbatim}

The gradient penalty uses interpolates between real and fake samples, with $\lambda = 10$. Note that we compute gradients efficiently by summing over the batch and taking derivatives w.r.t. the entire batch tensor.

\section{Ablations and discussion}
\subsection{Loss functions}
We observed (as expected) that the non-saturating generator loss improves early training dynamics compared to the original minimax generator objective. WGAN-GP stabilizes training but requires careful tuning of critic iterations and gradient penalty coefficient.

\subsection{Architectural choices}
The pixel shuffle upsampling used in the generator is a lightweight way to avoid checkerboard artifacts typical of naive transposed convolutions; batch normalization in the generator and careful use of leaky ReLU in the discriminator were important for stable learning.

\subsection{Failure modes and remedies}
Common failure modes and recommended mitigations:
\begin{itemize}
  \item Mode collapse: use minibatch discrimination, feature matching loss, or unrolled GAN techniques.
  \item Training instability: switch to WGAN-GP, tune learning rates, or use spectral normalization on the discriminator.
  \item Vanishing gradients for G: use the non-saturating loss or add feature matching objectives.
\end{itemize}
