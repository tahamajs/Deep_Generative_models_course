\section{مبانی نظری مدل‌های انتشار}

\subsection{فرآیند پیش‌رونده (Forward Process)}
در مدل‌های انتشار، فرآیند پیش‌رونده یک زنجیره مارکوف است که به تدریج به داده نویز اضافه می‌کند. طبق فرمول‌بندی \lr{Ho et al.}، توزیع شرطی در هر گام به صورت زیر است:
\begin{equation}
q(x_t | x_{t-1}) = \mathcal{N}(x_t; \sqrt{1-\beta_t}x_{t-1}, \beta_t I)
\end{equation}
با استفاده از خاصیت گاوسی، می‌توانیم $x_t$ را مستقیماً از $x_0$ نمونه‌برداری کنیم:
\begin{equation}
q(x_t | x_0) = \mathcal{N}(x_t; \sqrt{\bar{\alpha}_t}x_0, (1-\bar{\alpha}_t)I)
\end{equation}
که در آن $\alpha_t = 1 - \beta_t$ و $\bar{\alpha}_t = \prod_{s=1}^t \alpha_s$ است.

\subsection{فرآیند معکوس (Reverse Process)}
هدف مدل یادگیری فرآیند معکوس $p_\theta(x_{t-1}|x_t)$ است تا بتواند از نویز خالص $x_T \sim \mathcal{N}(0, I)$ به داده‌ی واقعی $x_0$ برسد. این توزیع نیز گاوسی در نظر گرفته می‌شود:
\begin{equation}
p_\theta(x_{t-1}|x_t) = \mathcal{N}(x_{t-1}; \mu_\theta(x_t, t), \Sigma_\theta(x_t, t))
\end{equation}
در عمل، شبکه عصبی معمولاً نویز $\epsilon_\theta(x_t, t)$ را پیش‌بینی می‌کند و میانگین $\mu_\theta$ از روی آن محاسبه می‌شود.

\subsection{تابع هدف (Objective Function)}
تابع هزینه ساده‌شده‌ای که برای آموزش استفاده می‌شود، میانگین مربعات خطا بین نویز واقعی $\epsilon$ و نویز پیش‌بینی‌شده توسط مدل است:
\begin{equation}
L_{\text{simple}}(\theta) = \mathbb{E}_{x_0, \epsilon, t} \left[ \| \epsilon - \epsilon_\theta(\sqrt{\bar{\alpha}_t}x_0 + \sqrt{1-\bar{\alpha}_t}\epsilon, t) \|^2 \right]
\end{equation}

\subsection{پرسش‌های نظری تکمیلی}

\subsubsection*{سوال ۱: اثبات فرم بسته‌ی $q(x_t|x_0)$}
برای اثبات رابطه‌ی
$q(x_t|x_0)=\mathcal{N}(\sqrt{\bar{\alpha}_t}x_0,(1-\bar{\alpha}_t)I)$
از کرنل تک‌گام استفاده می‌کنیم:
\begin{equation}
q(x_t|x_{t-1})=\mathcal{N}(x_t;\sqrt{\alpha_t}x_{t-1},(1-\alpha_t)I)
\end{equation}
پس می‌توان نمونه‌گیری را به صورت بازپارامترده نوشت:
\begin{equation}
x_t=\sqrt{\alpha_t}x_{t-1}+\sqrt{1-\alpha_t}\,\epsilon_{t-1},\quad \epsilon_{t-1}\sim\mathcal{N}(0,I)
\end{equation}

\section{جزئیات پیاده‌سازی و معماری‌ها}

\subsection{معماری U-Net برای DDPM}

\textbf{معماری کلی U-Net:}
\begin{itemize}
    \item \textbf{ورودی:} ۳۲×۳۲×۱ تصویر (MNIST با padding)
    \item \textbf{کانال‌ها:} [۶۴, ۱۲۸, ۲۵۶, ۵۱۲]
    \item \textbf{شرط‌گذاری زمانی:} embedding سینوسی → خطی → SiLU
    \item \textbf{توجه:} self-attention چندسر در bottleneck
\end{itemize}

\textbf{بلوک ResNet با شرط‌گذاری زمانی:}
\begin{minted}[fontsize=\footnotesize]{python}
class ResNetBlock(nn.Module):
    def __init__(self, in_channels, out_channels, time_emb_dim):
        super().__init__()
        self.conv1 = nn.Conv2d(in_channels, out_channels, 3, padding=1)
        self.conv2 = nn.Conv2d(out_channels, out_channels, 3, padding=1)
        self.time_emb = nn.Linear(time_emb_dim, out_channels)
        self.norm1 = nn.GroupNorm(32, in_channels)
        self.norm2 = nn.GroupNorm(32, out_channels)
        
        if in_channels != out_channels:
            self.shortcut = nn.Conv2d(in_channels, out_channels, 1)
        else:
            self.shortcut = nn.Identity()
    
    def forward(self, x, t_emb):
        h = self.norm1(x)
        h = F.silu(h)
        h = self.conv1(h)
        
        # Add time conditioning
        t_emb = self.time_emb(t_emb)
        t_emb = t_emb.unsqueeze(-1).unsqueeze(-1)  # [B, C, 1, 1]
        h = h + t_emb
        
        h = self.norm2(h)
        h = F.silu(h)
        h = self.conv2(h)
        
        return h + self.shortcut(x)
\end{minted}

\textbf{Embedding زمانی سینوسی:}
\begin{minted}[fontsize=\footnotesize]{python}
class SinusoidalEmbedding(nn.Module):
    def __init__(self, dim):
        super().__init__()
        self.dim = dim
    
    def forward(self, t):
        device = t.device
        half_dim = self.dim // 2
        emb = math.log(10000) / (half_dim - 1)
        emb = torch.exp(torch.arange(half_dim, device=device) * -emb)
        emb = t.unsqueeze(-1) * emb.unsqueeze(0)
        emb = torch.cat([torch.sin(emb), torch.cos(emb)], dim=-1)
        return emb
\end{minted}

\textbf{کلاس کامل U-Net:}
\begin{minted}[fontsize=\footnotesize]{python}
class UNet(nn.Module):
    def __init__(self, in_channels=3, time_emb_dim=256):
        super().__init__()
        
        # Time embedding
        self.time_emb = nn.Sequential(
            SinusoidalEmbedding(time_emb_dim),
            nn.Linear(time_emb_dim, time_emb_dim),
            nn.SiLU(),
            nn.Linear(time_emb_dim, time_emb_dim)
        )
        
        # Encoder
        self.enc1 = ResNetBlock(in_channels, 64, time_emb_dim)
        self.enc2 = ResNetBlock(64, 128, time_emb_dim)
        self.enc3 = ResNetBlock(128, 256, time_emb_dim)
        self.enc4 = ResNetBlock(256, 512, time_emb_dim)
        
        # Bottleneck with attention
        self.bottleneck = ResNetBlock(512, 512, time_emb_dim)
        self.attn = nn.MultiheadAttention(512, 8, batch_first=True)
        
        # Decoder
        self.dec4 = ResNetBlock(1024, 256, time_emb_dim)
        self.dec3 = ResNetBlock(512, 128, time_emb_dim)
        self.dec2 = ResNetBlock(256, 64, time_emb_dim)
        self.dec1 = ResNetBlock(128, 64, time_emb_dim)
        
        # Output
        self.final = nn.Conv2d(64, in_channels, 1)
        
        # Down/Up sampling
        self.down = nn.MaxPool2d(2)
        self.up = nn.Upsample(scale_factor=2, mode='bilinear')
    
    def forward(self, x, t):
        # Time embedding
        t_emb = self.time_emb(t)
        
        # Encoder
        e1 = self.enc1(x, t_emb)
        e2 = self.enc2(self.down(e1), t_emb)
        e3 = self.enc3(self.down(e2), t_emb)
        e4 = self.enc4(self.down(e3), t_emb)
        
        # Bottleneck
        b = self.bottleneck(self.down(e4), t_emb)
        b = b.flatten(2).transpose(1, 2)  # [B, H*W, C]
        b, _ = self.attn(b, b, b)
        b = b.transpose(1, 2).view(b.shape[0], -1, 4, 4)
        
        # Decoder
        d4 = self.dec4(torch.cat([self.up(b), e4], dim=1), t_emb)
        d3 = self.dec3(torch.cat([self.up(d4), e3], dim=1), t_emb)
        d2 = self.dec2(torch.cat([self.up(d3), e2], dim=1), t_emb)
        d1 = self.dec1(torch.cat([self.up(d2), e1], dim=1), t_emb)
        
        return self.final(d1)
\end{minted}

\subsection{DDIM Sampling}

\textbf{فرمول DDIM:}
برای $\eta = 0$ (تعیین‌گر):
\begin{equation}
x_{t-1} = \sqrt{\alpha_{t-1}} \left( \frac{x_t - \sqrt{1-\bar{\alpha}_t} \epsilon_\theta(x_t, t)}{\sqrt{\bar{\alpha}_t}} + \sqrt{1-\frac{\bar{\alpha}_{t-1}}{\bar{\alpha}_t}} \epsilon_\theta(x_t, t) \right)
\end{equation}

\textbf{پیاده‌سازی DDIM:}
\begin{minted}[fontsize=\footnotesize]{python}
def ddim_sample(self, model, x_t, t, t_prev, eta=0.0):
    """
    DDIM sampling step
    """
    # Predict noise
    epsilon = model(x_t, t)
    
    # Get alphas
    alpha_t = self.alphas_cumprod[t]
    alpha_t_prev = self.alphas_cumprod[t_prev] if t_prev >= 0 else 1.0
    
    # DDIM formulas
    pred_x0 = (x_t - torch.sqrt(1 - alpha_t) * epsilon) / torch.sqrt(alpha_t)
    
    dir_xt = torch.sqrt(1 - alpha_t_prev) * epsilon
    if eta > 0:
        # Add noise for stochasticity
        noise = torch.randn_like(x_t)
        dir_xt = dir_xt + eta * torch.sqrt((1 - alpha_t/alpha_t_prev) * (1 - alpha_t_prev)) * noise
    
    x_t_prev = torch.sqrt(alpha_t_prev) * pred_x0 + dir_xt
    
    return x_t_prev
\end{minted}

\subsection{Stable Diffusion با DreamBooth}

\textbf{معماری Stable Diffusion:}
\begin{itemize}

    \item \textbf{Embedding زمانی:} ۱۲۸۰-dim sinusoidal + linear projection
    \item \textbf{شرط‌گذاری متنی:} cross-attention با embeddings CLIP (۷۶۸-dim)
    \item \textbf{معماری:} مشابه U-Net DDPM اما با:
        \begin{itemize}
            \item بلوک‌های Transformer با self/cross-attention
            \item ۴ سر توجه در هر بلوک
            \item کانال‌ها: [۳۲۰, ۶۴۰, ۱۲۸۰, ۱۲۸۰]
        \end{itemize}
\end{itemize}

class DreamBoothDataset(Dataset):
    def __init__(self, instance_data_root, instance_prompt, tokenizer,
                 class_data_root=None, class_prompt=None, size=512):
        self.instance_data_root = Path(instance_data_root)
        self.instance_prompt = instance_prompt
        self.class_prompt = class_prompt
        self.tokenizer = tokenizer
        self.size = size
        
        # Load instance images
        self.instance_images = []
        for ext in ['*.jpg', '*.jpeg', '*.png']:
            self.instance_images.extend(list(self.instance_data_root.glob(ext)))
        
        # Load class images (for prior preservation)
        self.class_images = []
        if class_data_root:
            class_path = Path(class_data_root)
            for ext in ['*.jpg', '*.jpeg', '*.png']:
                self.class_images.extend(list(class_path.glob(ext)))
        
        # Image transforms
        self.transforms = transforms.Compose([
            transforms.Resize(size, interpolation=transforms.InterpolationMode.BILINEAR),
            transforms.CenterCrop(size),
            transforms.ToTensor(),
            transforms.Normalize([0.5], [0.5])
        ])
    
    def __len__(self):
        return max(len(self.instance_images), len(self.class_images))
    
    def __getitem__(self, idx):
        # Instance image
        instance_idx = idx % len(self.instance_images)
        instance_image = Image.open(self.instance_images[instance_idx]).convert("RGB")
        instance_pixel_values = self.transforms(instance_image)
        
        # Instance prompt tokenization
        instance_inputs = self.tokenizer(
            self.instance_prompt,
            padding="max_length",
            max_length=self.tokenizer.model_max_length,
            truncation=True,
            return_tensors="pt"
        )
        
        example = {
            "pixel_values": instance_pixel_values,
            "input_ids": instance_inputs.input_ids.squeeze()
        }
        
        # Class image (prior preservation)
        if self.class_images:
            class_idx = idx % len(self.class_images)
            class_image = Image.open(self.class_images[class_idx]).convert("RGB")
            class_pixel_values = self.transforms(class_image)
            
            class_inputs = self.tokenizer(
                self.class_prompt,
                padding="max_length",
                max_length=self.tokenizer.model_max_length,
                truncation=True,
                return_tensors="pt"
            )
            
            example.update({
                "class_pixel_values": class_pixel_values,
                "class_input_ids": class_inputs.input_ids.squeeze()
            })
        
        return example

\begin{minted}[fontsize=\footnotesize]{python}
class DreamBoothTrainer:
    def __init__(self, model_id="runwayml/stable-diffusion-v1-5", 
                 lora_rank=4, learning_rate=1e-4):
        # Load components
        self.tokenizer = CLIPTokenizer.from_pretrained(f"{model_id}/tokenizer")
        self.text_encoder = CLIPTextModel.from_pretrained(f"{model_id}/text_encoder")
        self.vae = AutoencoderKL.from_pretrained(f"{model_id}/vae")
        self.unet = UNet2DConditionModel.from_pretrained(f"{model_id}/unet")
        self.noise_scheduler = DDPMScheduler.from_pretrained(f"{model_id}/scheduler")
        
        # Freeze components
        self.vae.requires_grad_(False)
        self.text_encoder.requires_grad_(False)
        
        # Apply LoRA
        lora_config = LoraConfig(
            r=lora_rank,
            lora_alpha=lora_rank,
            target_modules=["to_k", "to_q", "to_v", "to_out.0"],
            init_lora_weights="gaussian"
        )
        self.unet = get_peft_model(self.unet, lora_config)
        
        # Optimizer (only LoRA parameters)
        self.optimizer = torch.optim.AdamW(
            self.unet.parameters(), lr=learning_rate, weight_decay=1e-2
        )
        
        self.scaler = torch.cuda.amp.GradScaler()
    
    def train_step(self, batch):
        pixel_values = batch["pixel_values"].to(self.device)
        input_ids = batch["input_ids"].to(self.device)
        
        # Encode to latents
        latents = self.vae.encode(pixel_values).latent_dist.sample() * 0.18215
        
        # Sample noise and timesteps
        noise = torch.randn_like(latents)
        timesteps = torch.randint(0, 1000, (latents.shape[0],), device=self.device)
        
        # Add noise
        noisy_latents = self.noise_scheduler.add_noise(latents, noise, timesteps)
        
        # Encode text
        encoder_hidden_states = self.text_encoder(input_ids)[0]
        
        # Predict noise
        with torch.cuda.amp.autocast():
            noise_pred = self.unet(noisy_latents, timesteps, encoder_hidden_states).sample
            loss = F.mse_loss(noise_pred, noise)
        
        # Backprop
        self.optimizer.zero_grad()
        self.scaler.scale(loss).backward()
        self.scaler.step(self.optimizer)
        self.scaler.update()
        
        return loss.item()
\end{minted}

\subsubsection*{سوال ۲: دو مزیت پیش‌بینی نویز $\epsilon$ به‌جای میانگین $\mu_\theta$}
\begin{enumerate}
    \item \textbf{پایداری عددی و مقیاس ثابتِ هدف:} نویز هدف همواره از \lr{$\mathcal{N}(0,I)$} می‌آید و مستقل از زمان $t$ مقیاس ثابتی دارد، در نتیجه یادگیری پایدارتر می‌شود.
    \item \textbf{کیفیت بهتر نمونه‌ها و ساده شدن وزن‌دهی VLB:} در فرمول‌بندی \lr{DDPM} پیش‌بینی نویز به یک هزینه‌ی \lr{MSE} ساده منجر می‌شود که از نظر تجربی اغلب کیفیت نمونه‌های بهتری نسبت به یادگیری مستقیم $\mu_\theta$ یا $x_0$ می‌دهد.
\end{enumerate}

\subsubsection*{سوال ۳: نقش ترم‌های KL در VLB و علت نادیده‌گرفتن در عمل}
	extbf{الف) تشویق به هم‌خوانی با پسین واقعی:}
ترم‌های \lr{KL} مانند
$D_{KL}(q(x_{t-1}|x_t,x_0)\,\|\,p_\theta(x_{t-1}|x_t))$
باعث می‌شوند فرآیند معکوس یادگرفته‌شده به پسین واقعی فرآیند پیش‌رونده نزدیک شود.

        extbf{ب) دلیل عملی:}
در بسیاری از پیاده‌سازی‌ها، به‌جای وزن‌دهی دقیق \lr{VLB} از هزینه‌ی ساده‌شده‌ی \lr{MSE} استفاده می‌شود چون (۱) بهینه‌سازی پایدارتر است و (۲) برای تولید تصویر، کیفیت ادراکی نمونه‌ها اهمیت بیشتری از بهینه‌سازی دقیق لگاریتم درست‌نمایی دارد.

\subsubsection*{سوال ۴: نمونه‌گیری DDIM}
	extbf{الف) تعیین‌گر شدن:}
در \lr{DDIM} با تنظیم پارامتر نویز تزریقی (معمولاً با نماد $\eta$ یا $\sigma_t$) روی صفر، جمله‌ی نویز حذف می‌شود و نگاشت $x_t\to x_{t-1}$ کاملاً تعیین‌گر می‌گردد.

        extbf{ب) سرعت بیشتر:}
چون مسیر تعیین‌گر است، می‌توان گام‌ها را با فواصل بزرگ‌تر برداشت (\lr{step skipping}) و به‌جای ۱۰۰۰ گام، با تعداد کمی گام (مثلاً ۵۰) نمونه‌های قابل قبول تولید کرد.

\subsubsection*{سوال ۵: Classifier-Free Guidance (CFG)}
	extbf{الف) ایده‌ی اصلی:}
در \lr{CFG} مدل هم‌زمان برای حالت شرطی و بی‌شرط آموزش داده می‌شود (با حذف تصادفی شرط در بخشی از داده‌ها). سپس در نمونه‌گیری، نویز نهایی به‌صورت ترکیبی از پیش‌بینی شرطی و بی‌شرط ساخته می‌شود:
\begin{equation}
        ilde{\epsilon}_\theta(x_t,c)=\epsilon_\theta(x_t|\emptyset)+w\big(\epsilon_\theta(x_t|c)-\epsilon_\theta(x_t|\emptyset)\big)
\end{equation}
که $w$ مقیاس هدایت است.

        extbf{ب) اثر $w$:}
افزایش $w$ معمولاً \textbf{وفاداری به پرامپت و شارپنس} را بیشتر می‌کند اما \textbf{تنوع} را کاهش می‌دهد و اگر خیلی بزرگ شود می‌تواند مصنوعات و اغراق در رنگ/بافت ایجاد کند.

\subsection{گزارش گام‌به‌گام پیاده‌سازی DDPM/DDIM (مطابق صورت سؤال)}
در این قسمت، مراحل اصلی پیاده‌سازی مطابق صورت سؤال ارائه می‌شود. (کد کامل در نوت‌بوک تمرین موجود است.)

\subsubsection*{گام ۱: پیاده‌سازی زمان‌بندی نویز (Noise Schedule)}
برای زمان‌بندی خطی، $\beta_t$ بین دو مقدار شروع و پایان به صورت خطی تغییر می‌کند و سپس $\alpha_t=1-\beta_t$ و $\bar{\alpha}_t$ پیش‌محاسبه می‌شوند.
\begin{minted}[fontsize=\footnotesize]{python}
class DDPMScheduler:
    def __init__(self, num_timesteps=1000, beta_start=1e-4, beta_end=2e-2, device="cuda"):
        self.num_timesteps = num_timesteps
        self.device = device

        self.betas = torch.linspace(beta_start, beta_end, num_timesteps, device=device)
        self.alphas = 1.0 - self.betas
        self.alphas_cumprod = torch.cumprod(self.alphas, dim=0)

        self.sqrt_alphas_cumprod = torch.sqrt(self.alphas_cumprod)
        self.sqrt_one_minus_alphas_cumprod = torch.sqrt(1.0 - self.alphas_cumprod)
\end{minted}

\subsubsection*{گام ۲: پیاده‌سازی \lr{perturb\_input} (فرآیند پیش‌رونده)}
با ترفند بازپارامترده:
$x_t=\sqrt{\bar{\alpha}_t}x_0+\sqrt{1-\bar{\alpha}_t}\epsilon$.
\begin{minted}[fontsize=\footnotesize]{python}
def perturb_input(self, x0, t, noise=None):
    if noise is None:
        noise = torch.randn_like(x0, device=self.device)
    sqrt_ab = self._extract(self.sqrt_alphas_cumprod, t, x0.shape)
    sqrt_1mab = self._extract(self.sqrt_one_minus_alphas_cumprod, t, x0.shape)
    xt = sqrt_ab * x0 + sqrt_1mab * noise
    return xt, noise
\end{minted}

\subsubsection*{گام ۳: حلقه آموزش (Training Loop)}
در هر گام آموزش، $t$ به‌صورت تصادفی نمونه‌برداری می‌شود، نویز افزوده می‌شود، مدل نویز را پیش‌بینی می‌کند و هزینه \lr{MSE} کمینه می‌شود.
\begin{minted}[fontsize=\footnotesize]{python}
def train_step(self, x0):
    self.model.train()
    b = x0.shape[0]
    t = torch.randint(0, self.scheduler.num_timesteps, (b,), device=self.device)
    xt, noise = self.scheduler.perturb_input(x0, t)
    noise_pred = self.model(xt, t)
    loss = F.mse_loss(noise_pred, noise)
    self.optimizer.zero_grad()
    loss.backward()
    torch.nn.utils.clip_grad_norm_(self.model.parameters(), 1.0)
    self.optimizer.step()
    return loss.item()
\end{minted}

\subsubsection*{گام ۴: نمونه‌گیری (Sampling) برای DDPM و DDIM}
نمونه‌گیری \lr{DDPM} از نویز گاوسی شروع می‌کند و با طی گام‌های معکوس، به تصویر می‌رسد. در \lr{DDIM} با $\eta=0$ فرآیند تعیین‌گر می‌شود و می‌توان تعداد گام‌ها را کم کرد.
\begin{minted}[fontsize=\footnotesize]{python}
@torch.no_grad()
def sample_ddpm(self, batch_size, img_size=(3, 32, 32)):
    self.model.eval()
    x = torch.randn(batch_size, *img_size, device=self.device)
    for t in reversed(range(self.scheduler.num_timesteps)):
        t_tensor = torch.full((batch_size,), t, device=self.device)
        eps = self.model(x, t_tensor)
        alpha = self.scheduler.alphas[t]
        alpha_bar = self.scheduler.alphas_cumprod[t]
        beta = self.scheduler.betas[t]
        z = torch.randn_like(x) if t > 0 else torch.zeros_like(x)
        x = (1.0 / torch.sqrt(alpha)) * (x - ((1 - alpha) / torch.sqrt(1 - alpha_bar)) * eps) + torch.sqrt(beta) * z
    return x
\end{minted}

\subsection{گزارش گام‌به‌گام DreamBooth (مطابق صورت سؤال)}
در این آزمایش از \lr{Stable Diffusion v1-5} استفاده شده و برای شخصی‌سازی سوژه، یک شناسه‌ی متنی ویژه (مثل \grayBox{sks}) به سوژه اختصاص داده می‌شود.

\subsubsection*{گام ۱: آماده‌سازی داده و پرامپت‌ها}
\begin{itemize}
    \item \textbf{Instance Prompt:} \lr{a photo of sks person}
    \item \textbf{Class Prompt:} \lr{a photo of a person}
\end{itemize}

\subsubsection*{گام ۲: کلاس دیتاست}
دیتاست باید هم تصاویر \lr{instance} و هم (در صورت استفاده) تصاویر \lr{class} برای \lr{prior preservation} را تولید کند.
\begin{minted}[fontsize=\footnotesize]{python}
def __getitem__(self, idx):
    example = {}
    instance_idx = idx % len(self.instance_images)
    instance_image = Image.open(self.instance_images[instance_idx]).convert("RGB")
    example["instance_images"] = self.transforms(instance_image)
    example["instance_prompt_ids"] = self.tokenizer(
        self.instance_prompt,
        truncation=True,
        padding="max_length",
        max_length=self.tokenizer.model_max_length,
        return_tensors="pt",
    ).input_ids.squeeze(0)

    if self.class_images is not None and len(self.class_images) > 0:
        class_idx = idx % len(self.class_images)
        class_image = Image.open(self.class_images[class_idx]).convert("RGB")
        example["class_images"] = self.transforms(class_image)
        example["class_prompt_ids"] = self.tokenizer(
            self.class_prompt,
            truncation=True,
            padding="max_length",
            max_length=self.tokenizer.model_max_length,
            return_tensors="pt",
        ).input_ids.squeeze(0)
    return example
\end{minted}

\subsubsection*{گام ۴: حلقه آموزش (LoRA Fine-tuning)}
برای سبک‌سازی آموزش، از \lr{LoRA} استفاده می‌شود تا تنها تعداد کمی پارامتر قابل آموزش باشد.
\begin{minted}[fontsize=\footnotesize]{python}
def train_step(self, batch):
    latents = self.vae.encode(batch["instance_images"].to(self.device)).latent_dist.sample() * 0.18215
    noise = torch.randn_like(latents)
    timesteps = torch.randint(0, self.noise_scheduler.config.num_train_timesteps, (latents.shape[0],), device=self.device).long()
    noisy_latents = self.noise_scheduler.add_noise(latents, noise, timesteps)
    encoder_hidden_states = self.text_encoder(batch["instance_prompt_ids"].to(self.device))[0]
    noise_pred = self.unet(noisy_latents, timesteps, encoder_hidden_states).sample
    loss = F.mse_loss(noise_pred, noise)
    return loss
\end{minted}
